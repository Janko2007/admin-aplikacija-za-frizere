  money(n) { return Number(n || 0).toLocaleString('sr-RS').replace(/,/g, '.'); }
  priceOf(s) {
    if (s.eur) return s.max ? s.price + '—' + s.max + ' €' : s.price + ' €';
    return s.max ? this.money(s.price) + '—' + this.money(s.max) + ' RSD' : this.money(s.price) + ' RSD';
  }

  tabStyle(on) {
    return on
      ? 'padding:9px 17px;border-radius:999px;border:1px solid rgba(227,176,196,.42);background:rgba(227,176,196,.16);color:#F4DCE5;font-size:12.5px;font-weight:600;cursor:pointer'
      : 'padding:9px 17px;border-radius:999px;border:1px solid rgba(255,255,255,.1);background:rgba(255,255,255,.035);color:rgba(242,236,239,.62);font-size:12.5px;font-weight:500;cursor:pointer';
  }

  navLinkStyle(on) {
    return 'position:relative;z-index:2;padding:9px 10px;border-radius:11px;white-space:nowrap;' +
      'transition:color .34s cubic-bezier(.22,1,.36,1);' +
      (on ? 'color:#F7E4EC;' : 'color:rgba(242,236,239,.76);');
  }

  send = () => {
    if (this.state.sent) return;
    this.setState({ sent: true });
    this._t = setTimeout(() => this.setState({ sent: false, fName: '', fMail: '', fMsg: '' }), 2400);
  };
  componentWillUnmount() { clearTimeout(this._t); }

  renderVals() {
    const cat = this.state.cat;
    const showDur = this.props.showDurations ?? true;
    const tabs = ['Sve'].concat(this.cats).map(c => ({
      label: c === 'Sve' ? 'Sve usluge' : (this.labels[c] || c),
      style: this.tabStyle(cat === c),
      pick: () => this.setState({ cat: c })
    }));
    const shown = cat === 'Sve' ? this.cats : [cat];
    const groups = shown.map(c => {
      const items = this.catalog.filter(s => s.cat === c).map(s => ({
        name: s.name, price: this.priceOf(s), dur: showDur ? s.dur + ' min' : ''
      }));
      return { cat: this.labels[c] || c, count: items.length + ' usluga', items };
    });
    const catCards = this.cats.map(c => {
      const items = this.catalog.filter(s => s.cat === c);
      const min = Math.min.apply(null, items.map(s => s.price));
      const cheapest = items.find(s => s.price === min);
      return {
        label: this.labels[c] || c,
        count: items.length + (items.length === 1 ? ' usluga' : ' usluga'),
        from: 'od ' + (cheapest.eur ? min + ' €' : this.money(min) + ' RSD'),
        open: () => this.setState({ cat: c, priceOpen: true })
      };
    });
    const p = this.state.pill;
    const moving = this.state.moving;
    const navItems = this.nav.map(n => ({
      label: n.label, href: '#' + n.id,
      style: this.navLinkStyle(this.state.active === n.id),
      ref: el => { if (el) this.linkEls[n.id] = el; },
      pick: () => this.setState({ active: n.id })
    }));
    return {
      navItems,
      pillStyle: p
        ? 'position:absolute;top:-3px;bottom:-3px;z-index:1;pointer-events:none;left:' + p.left + 'px;width:' + p.width + 'px;' +
          (p.flow
            ? 'transform:scaleX(' + (1 + p.v * 0.2) + ') scaleY(' + (1 - p.v * 0.14) + ');' +
              'transition:transform .3s cubic-bezier(.22,1,.36,1);will-change:left,width,transform;'
            : 'transform:scaleX(' + (moving ? 1.14 : 1) + ') scaleY(' + (moving ? 0.9 : 1) + ');' +
              'transition:left .62s cubic-bezier(.26,1.42,.3,1),width .62s cubic-bezier(.26,1.42,.3,1),transform .42s cubic-bezier(.26,1.42,.3,1);will-change:left,width,transform;')
        : 'display:none',
      pillInner: 'position:absolute;inset:0;border-radius:13px;display:block;overflow:hidden;' +
        'background:linear-gradient(150deg,rgba(255,255,255,.2),rgba(227,176,196,.16) 46%,rgba(255,255,255,.05));' +
        'border:1px solid rgba(255,255,255,.26);box-shadow:0 6px 18px rgba(0,0,0,.45),inset 0 1px 0 rgba(255,255,255,.4),inset 0 -6px 12px rgba(227,176,196,.2);' +
        'backdrop-filter:blur(9px) saturate(180%);-webkit-backdrop-filter:blur(9px) saturate(180%);',
      pillGlare: 'position:absolute;left:8px;right:8px;top:2px;height:42%;border-radius:11px 11px 18px 18px;display:block;' +
        'background:linear-gradient(180deg,rgba(255,255,255,.36),rgba(255,255,255,0));',
      tabs, groups, catCards,
      priceOpen: this.state.priceOpen,
      modalTitle: cat === 'Sve' ? 'Cene tretmana' : (this.labels[cat] || cat),
      openAll: () => this.setState({ cat: 'Sve', priceOpen: true }),
      closePrices: () => this.setState({ priceOpen: false }),
      partners: this.partners,
      showPartners: this.props.showPartners ?? true,
      showMap: this.props.showMap ?? true,
      fName: this.state.fName, fMail: this.state.fMail, fMsg: this.state.fMsg,
      setName: e => this.setState({ fName: e.target.value }),
      setMail: e => this.setState({ fMail: e.target.value }),
      setMsg: e => this.setState({ fMsg: e.target.value }),
      send: this.send,
      sendLabel: this.state.sent ? 'Upit je poslat ✓' : 'Pošalji'
    };
  }
